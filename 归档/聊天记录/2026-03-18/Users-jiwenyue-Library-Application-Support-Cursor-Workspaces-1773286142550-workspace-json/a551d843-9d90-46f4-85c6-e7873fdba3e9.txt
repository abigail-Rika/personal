# 工作日 10:40 自动提醒今日待办

TODOS_FILE="$HOME/cursorProjects/personal/work/todos.md"

# 只在工作日运行（1=周一 ... 5=周五）
DOW=$(date +%u)
if [ "$DOW" -gt 5 ]; then
  exit 0
fi

if [ ! -f "$TODOS_FILE" ]; then
  osascript -e 'display notification "todos.md 文件不存在" with title "⚠️ 工作提醒"'
  exit 1
fi

# 解析紧急重要区块中的任务标题和状态
IN_URGENT=0
ITEMS=""
COUNT=0

while IFS= read -r line; do
  # 进入"紧急重要"区块
  if echo "$line" | grep -q "^## 紧急重要"; then
    IN_URGENT=1
    continue
  fi
  # 遇到下一个二级标题则退出
  if [ "$IN_URGENT" -eq 1 ] && echo "$line" | grep -q "^## " ; then
    break
  fi
  # 提取任务标题行（### 开头）
  if [ "$IN_URGENT" -eq 1 ] && echo "$line" | grep -q "^### "; then
    TITLE=$(echo "$line" | sed 's/^### [0-9]*\. //')
    COUNT=$((COUNT + 1))
    # 读后续行找状态
    STATUS=""
  fi
  # 提取状态
  if [ "$IN_URGENT" -eq 1 ] && echo "$line" | grep -q "^\- \*\*状态\*\*"; then
    STATUS=$(echo "$line" | sed 's/.*：//')
    ITEMS="${ITEMS}${COUNT}. ${TITLE} ${STATUS}\n"
  fi
  # 提取验收/上线日期（附在标题后）
  if [ "$IN_URGENT" -eq 1 ] && echo "$line" | grep -q "^\- \*\*验收\*\*"; then
    DEADLINE=$(echo "$line" | sed 's/.*：//')
    ITEMS="${ITEMS}   验收: ${DEADLINE}\n"
  fi
done < "$TODOS_FILE"

TODAY=$(date '+%m月%d日 %A' | sed \
  -e 's/Monday/周一/' \
  -e 's/Tuesday/周二/' \
  -e 's/Wednesday/周三/' \
  -e 's/Thursday/周四/' \
  -e 's/Friday/周五/' \
  -e 's/Saturday/周六/' \
  -e 's/Sunday/周日/')

# 构建通知内容（macOS 通知有长度限制，保持精简）
if [ "$COUNT" -eq 0 ]; then
  SUMMARY="暂无紧急待办，保持节奏 💪"
else
  SUMMARY=$(printf "$ITEMS" | head -10)
fi

# 发送 macOS 通知
osascript -e "display notification \"${SUMMARY}\" with title \"📋 今日待办 · ${TODAY}\" subtitle \"共 ${COUNT} 项紧急任务\""

# 同时输出到终端（方便日志查看）
echo "=== 工作日报提醒 ${TODAY} ==="
printf "$ITEMS"
echo "=========================="

    LINE="${COUNT}. ${TITLE}"

  STATUS=""

  fi
  if [ "$IN_URGENT" -eq 1 ] && echo "$line" | grep -q "^\- \*\*状态\*\*"; then

  fi
done < "$TODOS_FILE"
